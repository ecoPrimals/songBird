pub async fn handle_sha3_256(params: Value) -> Result<Value> {
    let data_b64 = params["data"].as_str()?;
    let data = base64::decode(data_b64)?;
    let hash = Sha3_256::digest(&data);
    Ok(json!({ "hash": base64::encode(&hash) }))
}
