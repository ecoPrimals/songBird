```bash

cargo check --workspace || exit 1
cargo fmt --check || exit 1  
cargo clippy -- -D warnings || exit 1
EOF
chmod +x .git/hooks/pre-commit

# 2. GitHub Actions CI (future)
# 3. Automated backups with validation
```
